# List of Test_Batched_*.hpp groups that have been split out of the
# Test_Batched_Dense.hpp umbrella into their own translation units.  For each
# entry <GROUP>:
#   1. Test_Batched_<GROUP>.hpp must exist (its Real/Complex variants are
#      auto-detected below), and its #include line(s) must have been removed
#      from Test_Batched_Dense.hpp.
#   2. The per-backend shim .cpp files are auto-generated at configure time
#      into ${CMAKE_CURRENT_BINARY_DIR}/generated_backends/.
#   3. The group name is listed here.
#
# Every split .cpp file is added to the SAME executable as the corresponding
# umbrella .cpp, so `ctest` behaviour is unchanged; the split simply lets
# `make -j` compile the group in parallel with the umbrella TU.
#
# To split another group in the future: remove its includes from
# Test_Batched_Dense.hpp and append its stem here.  No other changes needed.
set(BATCHED_DLA_SPLIT_GROUPS SerialTrsm SerialTrmm SerialSolveLU)

# Generate one shim .cpp per (BACKEND, GROUP) into ${CMAKE_CURRENT_BINARY_DIR}
# and return the list in OUT_VAR.  Each shim mimics the hand-written pattern
# used by the umbrella backends/*.cpp files: an include guard, the backend
# header, and every existing Test_Batched_<GROUP>{,_Real,_Complex}.hpp header.
function(_kokkoskernels_batched_dla_split_sources BACKEND OUT_VAR)
  set(_srcs "")
  set(_outdir "${CMAKE_CURRENT_BINARY_DIR}/generated_backends")
  file(MAKE_DIRECTORY "${_outdir}")
  string(TOUPPER "${BACKEND}" _backend_upper)
  foreach(_group IN LISTS BATCHED_DLA_SPLIT_GROUPS)
    string(TOUPPER "${_group}" _group_upper)
    set(_guard "TEST_${_backend_upper}_BATCHED_${_group_upper}_CPP")
    set(_content
"// SPDX-License-Identifier: Apache-2.0 WITH LLVM-exception\n\
// SPDX-FileCopyrightText: Copyright Contributors to the Kokkos project\n\
//\n\
// AUTO-GENERATED by batched/dense/unit_test/CMakeLists.txt.  Do not edit.\n\
// Regenerated on every CMake configure from BATCHED_DLA_SPLIT_GROUPS.\n\
#ifndef ${_guard}\n\
#define ${_guard}\n\
\n\
#include <Test_${BACKEND}.hpp>\n\
#include \"Test_Batched_${_group}.hpp\"\n")
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Test_Batched_${_group}_Real.hpp")
      string(APPEND _content "#include \"Test_Batched_${_group}_Real.hpp\"\n")
    endif()
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Test_Batched_${_group}_Complex.hpp")
      string(APPEND _content "#include \"Test_Batched_${_group}_Complex.hpp\"\n")
    endif()
    string(APPEND _content "\n#endif  // ${_guard}\n")
    set(_out "${_outdir}/Test_${BACKEND}_Batched_${_group}.cpp")
    # file(CONFIGURE) writes only when content changes, preserving ccache
    # hits and avoiding needless rebuilds.
    file(CONFIGURE OUTPUT "${_out}" CONTENT "${_content}" @ONLY)
    list(APPEND _srcs "${_out}")
  endforeach()
  set(${OUT_VAR} "${_srcs}" PARENT_SCOPE)
endfunction()
