# List of Test_Batched_*.hpp groups that have been split out of the
# Test_Batched_Dense.hpp umbrella into their own translation units.  For each
# entry <GROUP>:
#   1. Test_Batched_<GROUP>.hpp must exist (its Real/Complex variants are
#      auto-detected below), and its #include line(s) must have been removed
#      from Test_Batched_Dense.hpp.
#   2. The per-backend shim .cpp files are auto-generated at configure time
#      into ${CMAKE_CURRENT_BINARY_DIR}/generated_backends/.
#   3. The group name is listed below in set(BATCHED_DLA_SPLIT_GROUPS ... <GROUP>).
#
# Every split .cpp file is added to the SAME executable as the corresponding
# umbrella .cpp, so `ctest` behaviour is unchanged; the split simply lets
# `make -j` compile the group in parallel with the umbrella TU.
#
# To split another group in the future: remove its includes from
# Test_Batched_Dense.hpp and append its stem here.  No other changes needed.

# Tests that only work on host space and should be excluded from GPU backends
set(BATCHED_DLA_HOST_ONLY_GROUPS
  VectorArithmatic
  VectorLogical
  VectorMath
  VectorMisc
  VectorRelation
  VectorView)

# Generate one shim .cpp per (BACKEND, GROUP) into ${CMAKE_CURRENT_BINARY_DIR}
# and return the list in OUT_VAR.  Each shim mimics the hand-written pattern
# used by the umbrella backends/*.cpp files: an include guard, the backend
# header, and every existing Test_Batched_<GROUP>{,_Real,_Complex}.hpp header.
function(kokkoskernels_batched_dla_split_sources BACKEND OUT_VAR)
  set(kkbdss_srcs "") # prefix vars with kkbdss to ensure unique
  set(kkbdss_outdir "${CMAKE_CURRENT_BINARY_DIR}/generated_backends")
  file(MAKE_DIRECTORY "${kkbdss_outdir}")
  string(TOUPPER "${BACKEND}" kkbdss_backend_upper)
  foreach(kkbdss_group IN LISTS BATCHED_DLA_SPLIT_GROUPS)
    # Skip host-only tests on GPU backends
    if(BACKEND MATCHES "Cuda|HIP|SYCL|OpenMPTarget")
      if(kkbdss_group IN_LIST BATCHED_DLA_HOST_ONLY_GROUPS)
        continue()
      endif()
    endif()
    string(TOUPPER "${kkbdss_group}" kkbdss_group_upper)
    set(kkbdss_guard "TEST_${kkbdss_backend_upper}_BATCHED_${kkbdss_group_upper}_CPP")
    set(kkbdss_content
"// SPDX-License-Identifier: Apache-2.0 WITH LLVM-exception\n\
// SPDX-FileCopyrightText: Copyright Contributors to the Kokkos project\n\
//\n\
// AUTO-GENERATED by batched/dense/unit_test/CMakeLists.txt.  Do not edit.\n\
// Regenerated on every CMake configure from BATCHED_DLA_SPLIT_GROUPS.\n\
#ifndef ${kkbdss_guard}\n\
#define ${kkbdss_guard}\n\
\n\
#include <Test_${BACKEND}.hpp>\n\
#include \"Test_Batched_${kkbdss_group}.hpp\"\n")
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Test_Batched_${kkbdss_group}_Real.hpp")
      string(APPEND kkbdss_content "#include \"Test_Batched_${kkbdss_group}_Real.hpp\"\n")
    endif()
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Test_Batched_${kkbdss_group}_Complex.hpp")
      string(APPEND kkbdss_content "#include \"Test_Batched_${kkbdss_group}_Complex.hpp\"\n")
    endif()
    string(APPEND kkbdss_content "\n#endif  // ${kkbdss_guard}\n")
    set(kkbdss_out "${kkbdss_outdir}/Test_${BACKEND}_Batched_${kkbdss_group}.cpp")
    # file(GENERATE) writes only when content changes, preserving ccache
    # hits and avoiding needless rebuilds.
    file(GENERATE OUTPUT "${kkbdss_out}" CONTENT "${kkbdss_content}")
    list(APPEND kkbdss_srcs "${kkbdss_out}")
  endforeach()
  set(${OUT_VAR} "${kkbdss_srcs}" PARENT_SCOPE)
endfunction()
